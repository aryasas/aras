YAML Format Reference
app:
  name: inventory
  title: Inventory
  url: /inventory

tables:
  - name: products
    title: Products
    url_suffix: /products
    columns:
      - name: name
        label: Name
        field_type: string
        required: true
      - name: price
        label: Price
        field_type: decimal
      - name: category
        label: Category
        field_type: select
        choices: "Electronics,Clothing,Food"
