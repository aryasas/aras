import { Tabs } from 'expo-router'
import { Chrome, FileText, Plus } from 'lucide-react-native'

export default function TabLayout() {
  return (
    <Tabs
      screenOptions={{
        tabBarActiveTintColor: '#6366F1',
        tabBarStyle: { backgroundColor: '#FFFFFF' },
        headerShown: false,
      }}
    >
      <Tabs.Screen name="index"
        options={{ title: 'Home', tabBarIcon: ({ color }) => <Chrome color={color} size={22} /> }}
      />
      <Tabs.Screen name="invoices"
        options={{ title: 'Invoices', tabBarIcon: ({ color }) => <FileText color={color} size={22} /> }}
      />
      <Tabs.Screen name="new"
        options={{
          title: 'New',

          ),
        }}
      />
    </Tabs>
  )

✓ Expo Router v3 uses file-based routing. The (tabs) folder = tab group. Each file inside = one tab. The custom FAB-style center tab uses a View override for the icon. ⚠ lucide-react-native requires react-native-svg: npx expo install react-native-svg.
